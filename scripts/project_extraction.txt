import sys
import hashlib
from pathlib import Path
from collections import defaultdict
import numpy as np
from PIL import Image

def get_md5(file_path):
    hash_md5 = hashlib.md5()
    with open(file_path, "rb") as f:
        for chunk in iter(lambda: f.read(4096), b""):
            hash_md5.update(chunk)
    return hash_md5.hexdigest()

def get_v2_training_hashes(v2_dir):
    v2_hashes = set()
    v2_path = Path(v2_dir)
    if not v2_path.exists():
        print(f"Error: {v2_dir} does not exist.")
        return v2_hashes
        
    for ext in ["*.jpg", "*.jpeg", "*.png", "*.webp"]:
        for f in v2_path.rglob(ext):
            v2_hashes.add(get_md5(f))
    return v2_hashes

def audit_directory(directory_path, v2_hashes, skip_dirs):
    dir_path = Path(directory_path)
    if not dir_path.exists() or not dir_path.is_dir():
        return None
        
    # Skip excluded dirs
    for skip in skip_dirs:
        if skip in str(dir_path.resolve()):
            return None
            
    image_files = []
    extensions = set()
    for ext in ["*.jpg", "*.jpeg", "*.png", "*.webp"]:
        for f in dir_path.rglob(ext):
            # Check skip dirs again per file just to be safe
            skip_file = False
            for skip in skip_dirs:
                if skip in str(f.resolve()):
                    skip_file = True
                    break
            if not skip_file:
                image_files.append(f)
                extensions.add(f.suffix.lower())
                
    if not image_files:
        return None
        
    widths = []
    heights = []
    aspect_ratios = []
    landscape = 0
    portrait = 0
    square = 0
    
    exact_duplicates = 0
    
    for f in image_files:
        if get_md5(f) in v2_hashes:
            exact_duplicates += 1
            
        try:
            with Image.open(f) as img:
                w, h = img.size
                widths.append(w)
                heights.append(h)
                ar = w / h
                aspect_ratios.append(ar)
                
                if ar > 1.05:
                    landscape += 1
                elif ar < 0.95:
                    portrait += 1
                else:
                    square += 1
        except Exception:
            pass
            
    stats = {
        "path": str(dir_path),
        "count": len(image_files),
        "extensions": list(extensions),
        "mean_w": np.mean(widths) if widths else 0,
        "mean_h": np.mean(heights) if heights else 0,
        "median_w": np.median(widths) if widths else 0,
        "median_h": np.median(heights) if heights else 0,
        "landscape": landscape,
        "portrait": portrait,
        "square": square,
        "duplicates": exact_duplicates
    }
    
    return stats

def main():
    print("=" * 80)
    print("PHASE 4B: EXISTING REAL-WORLD TUR DATA INVENTORY")
    print("=" * 80)
    
    v2_dir = "datasets/tur_image_validator_v2/train/Tur"

    v2_hashes = get_v2_training_hashes(v2_dir)
    print(f"Loaded {len(v2_hashes)} unique V2 hashes.\n")
    
    skip_dirs = [
        str(Path("datasets/tur_image_validator_v2/train/Tur").resolve()),
        str(Path("datasets/tur_image_validator/real_world_test").resolve()),
        str(Path("PPLD_extracted/PPLD_backup_before_duplicate_removal").resolve())
    ]
    
    candidate_dirs = [
        "datasets/new_disease_images",
        "datasets/tur",
        "datasets/tur_varieties",
        "datasets/external_pigeonpea_field",
        "datasets/pigeonpea_disease_final",

    ]
    
    print(f"{'Directory':<45} | {'Count':<6} | {'Ext':<10} | {'Shape (L/P/S)':<15} | {'Mean WxH':<15} | {'Dups':<5}")
    print("-" * 110)
    
    for cdir in candidate_dirs:

            print(f"{stats['path']:<45} | {stats['count']:<6} | {exts:<10} | {shape:<15} | {mean_dim:<15} | {stats['duplicates']:<5}")
            
    print("\nAudit complete.")

def get_file_sha256(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        while chunk := f.read(65536):
            h.update(chunk)
    return h.hexdigest()

def get_decoded_pixel_sha256(img):
    rgb_img = img.convert("RGB")
    return hashlib.sha256(rgb_img.tobytes()).hexdigest()

def compute_dhash(img, hash_size=8):
    resized = img.convert("L").resize((hash_size + 1, hash_size), Image.Resampling.LANCZOS)
    arr = np.array(resized)
    diff = arr[:, 1:] > arr[:, :-1]
    packed = np.packbits(diff.flatten())
    # Return as uint64
    return np.frombuffer(packed.tobytes(), dtype=">u8")[0]

def compute_phash(img, hash_size=8, highfreq_factor=4):
    img_size = hash_size * highfreq_factor
    resized = img.convert("L").resize((img_size, img_size), Image.Resampling.LANCZOS)
    arr = np.array(resized, dtype=np.float32)
    dct = cv2.dct(arr)
    dctlowfreq = dct[:hash_size, :hash_size]
    med = np.median(dctlowfreq[1:, 1:])
    diff = dctlowfreq > med
    packed = np.packbits(diff.flatten())
    # Return as uint64
    return np.frombuffer(packed.tobytes(), dtype=">u8")[0]

def popcount64(v):
    # Vectorized 64-bit Hamming weight
    v = v - ((v >> np.uint64(1)) & np.uint64(0x5555555555555555))
    v = (v & np.uint64(0x3333333333333333)) + ((v >> np.uint64(2)) & np.uint64(0x3333333333333333))
    v = (v + (v >> np.uint64(4))) & np.uint64(0x0f0f0f0f0f0f0f0f)
    return (v * np.uint64(0x0101010101010101)) >> np.uint64(56)

def index_upstream_sources():
    print("Indexing upstream provenance sources...")
    upstream_index = {}
    sources = [
        ("PPLD_extracted", PROJECT_ROOT / "PPLD_extracted"),
        ("PPLD_processed", PROJECT_ROOT / "PPLD_processed"),
        ("temp_wilt_staging", PROJECT_ROOT / "datasets" / "_temp_wilt_staging"),
        ("new_disease_images", PROJECT_ROOT / "datasets" / "new_disease_images"),
    ]
    for source_name, source_path in sources:
        if not source_path.exists():
            continue
        for f in source_path.rglob("*"):
            if f.is_file() and f.suffix.lower() in VALID_EXTENSIONS:
                try:
                    h = get_file_sha256(f)
                    rel_p = str(f.relative_to(PROJECT_ROOT))
                    upstream_index.setdefault(h, []).append({
                        "source": source_name,
                        "path": rel_p,
                        "parent": f.parent.name,
                        "filename": f.name
                    })
                except Exception as e:
                    pass
    print(f"Indexed {len(upstream_index)} unique file hashes across upstream sources.")
    return upstream_index

        rel_path = str(file_path.relative_to(DATASET_ROOT))

        pre_audit_manifest[rel_path] = {
            "size": file_size,
            "sha256": file_sha256,

        ext = file_path.suffix.lower()

        error_detail = ""
        width, height = 0, 0
        img_format = ""
        img_mode = ""

        pixel_sha256 = ""
        dh = np.uint64(0)
        ph = np.uint64(0)

        # Size check
        if file_size == 0:
            is_corrupt = True
            error_detail = "ZERO_BYTE_FILE"

        # Image integrity check
        if not is_corrupt:
            try:
                # 1. Structural verify
                with Image.open(file_path) as img:
                    img_format = img.format or ""
                    img_mode = img.mode or ""
                    width, height = img.size
                    img.verify()
                
                # Format vs extension mismatch check
                ext_clean = ext.lstrip(".")
                if ext_clean in ["jpg", "jpeg"] and img_format.lower() not in ["jpeg", "jpg"]:
                    error_detail = f"FORMAT_MISMATCH: ext={ext} but actual format={img_format}"
                elif ext_clean == "png" and img_format.lower() != "png":
                    error_detail = f"FORMAT_MISMATCH: ext={ext} but actual format={img_format}"

                # 2. Pixel load and decode
                with Image.open(file_path) as img:
                    img.load()
                    aspect_ratio = round(width / max(height, 1), 4)
                    pixel_sha256 = get_decoded_pixel_sha256(img)
                    dh = compute_dhash(img)
                    ph = compute_phash(img)

                # Channels / mode anomalies
                if img_mode not in ["RGB", "L"]:
                    if not error_detail:
                        error_detail = f"UNUSUAL_MODE: {img_mode}"

                if width < 32 or height < 32:
                    error_detail = f"EXTREME_SMALL_DIMENSIONS: {width}x{height}"

    # 5. Near Duplicate Detection using Vectorized Hamming Distance on pHash and dHash
    print("\n[Step 4/6] Computing perceptual similarity matrix (pHash & dHash)...")
    N = len(file_records)
    ph_array = np.array(phash_list, dtype=np.uint64)
    dh_array = np.array(dhash_list, dtype=np.uint64)

    # Compute pairwise XOR and popcount in chunks to minimize peak memory
    near_dup_pairs = []
    chunk_size = 500

    for i_start in range(0, N, chunk_size):
        i_end = min(i_start + chunk_size, N)
        ph_chunk = ph_array[i_start:i_end]
        dh_chunk = dh_array[i_start:i_end]

        # Pairwise distance matrices for chunk against all
        ph_xor = ph_chunk[:, None] ^ ph_array[None, :]
        dh_xor = dh_chunk[:, None] ^ dh_array[None, :]

        ph_dist = popcount64(ph_xor)
        dh_dist = popcount64(dh_xor)

        for i_rel in range(i_end - i_start):
            i = i_start + i_rel
            # Only consider j > i
            js = np.where((ph_dist[i_rel] <= 6) & (dh_dist[i_rel] <= 8))[0]
            for j in js:
                if j <= i:
                    continue
                
                r1 = file_records[i]
                r2 = file_records[j]

                # Skip exact duplicates already tracked
                if r1["file_sha256"] == r2["file_sha256"]:
                    continue

                p_d = int(ph_dist[i_rel, j])
                d_d = int(dh_dist[i_rel, j])

                near_dup_pairs.append({
                    "path_1": r1["relative_path"],
                    "split_1": r1["split"],
                    "class_1": r1["assigned_class"],
                    "path_2": r2["relative_path"],
                    "split_2": r2["split"],
                    "class_2": r2["assigned_class"],
                    "phash_dist": p_d,
                    "dhash_dist": d_d,
                    "is_cross_split": r1["split"] != r2["split"],

                    "prov_1": r1["prov_original_filenames"],
                    "prov_2": r2["prov_original_filenames"],
                })

    print(f"Near-duplicate pairs detected (pHash dist <= 6, dHash dist <= 8, non-exact): {len(near_dup_pairs)}")

    # Classify near-duplicate cross-split leakage and cross-class conflicts
    near_leakage_pairs = [p for p in near_dup_pairs if p["is_cross_split"]]
    near_cross_class_pairs = [p for p in near_dup_pairs if p["is_cross_class"]]

    print(f"  - Cross-split near-duplicate leakage pairs : {len(near_leakage_pairs)}")
    print(f"  - Cross-class near-duplicate conflict pairs : {len(near_cross_class_pairs)}")

    # 6. Label Review Manifest Assembly
    print("\n[Step 5/6] Building Label Quality & Wrong-Crop Review Manifest...")
    label_review_manifest = []

    # A. Exact conflicts
    for conf in cross_class_exact_conflicts:
        for p in conf["paths"]:
            rec = next(r for r in file_records if r["relative_path"] == p)
            label_review_manifest.append({
                "relative_path": rec["relative_path"],
                "split": rec["split"],
                "assigned_class": rec["assigned_class"],
                "review_category": "CONFIRMED_CROSS_CLASS_DUPLICATE",
                "severity": "CRITICAL",
                "confidence": "CONFIRMED",
                "evidence": f"Exact byte match across classes: {conf['classes_involved']}. Upstream prov: {rec['prov_original_classes']}",
                "upstream_source": rec["prov_sources"],
                "upstream_filenames": rec["prov_original_filenames"],
                "recommended_action": "EXCLUDE_OR_RESOLVE_LABEL"
            })

    # B. Near-duplicate cross-class conflicts (dist <= 4)
    seen_near_conflict_paths = set()
    for p in near_cross_class_pairs:
        if p["phash_dist"] <= 4:
            for path_key, split_key, cls_key, opp_cls in [
                ("path_1", "split_1", "class_1", p["class_2"]),
                ("path_2", "split_2", "class_2", p["class_1"])
            ]:
                pth = p[path_key]
                if pth not in seen_near_conflict_paths:
                    seen_near_conflict_paths.add(pth)
                    rec = next(r for r in file_records if r["relative_path"] == pth)
                    label_review_manifest.append({
                        "relative_path": rec["relative_path"],
                        "split": rec["split"],
                        "assigned_class": rec["assigned_class"],
                        "review_category": "SUSPECTED_CROSS_CLASS_NEAR_DUPLICATE",
                        "severity": "HIGH",
                        "confidence": "SUSPECTED",
                        "evidence": f"Perceptually near-identical (phash_dist={p['phash_dist']}, dhash_dist={p['dhash_dist']}) to image in {opp_cls}",

import sys

def get_md5(file_path):
    hash_md5 = hashlib.md5()
    with open(file_path, "rb") as f:
        for chunk in iter(lambda: f.read(4096), b""):
            hash_md5.update(chunk)
    return hash_md5.hexdigest()

def get_hashes_from_dir(dir_path_str):
    hashes = set()
    p = Path(dir_path_str)
    if not p.exists():
        return hashes
    for ext in ["*.jpg", "*.jpeg", "*.png", "*.webp"]:
        for f in p.rglob(ext):
            hashes.add(get_md5(f))
    return hashes

def create_contact_sheet(image_paths, output_path, cols=10, thumb_size=128):
    """Creates a simple visual contact sheet without modifying original images."""
    if not image_paths:
        return
        
    # Calculate rows
    rows = (len(image_paths) + cols - 1) // cols
    
    # Create blank canvas
    sheet_width = cols * thumb_size
    sheet_height = rows * thumb_size
    sheet = Image.new("RGB", (sheet_width, sheet_height), color=(255, 255, 255))
    
    for i, img_path in enumerate(image_paths):
        try:
            with Image.open(img_path) as img:
                img_copy = img.copy()
                img_copy.thumbnail((thumb_size, thumb_size))
                
                # Center the thumbnail in its cell
                x = (i % cols) * thumb_size
                y = (i // cols) * thumb_size
                
                # Calculate offsets for centering
                w, h = img_copy.size
                offset_x = (thumb_size - w) // 2
                offset_y = (thumb_size - h) // 2
                
                sheet.paste(img_copy, (x + offset_x, y + offset_y))
        except Exception:
            pass
            
    sheet.save(output_path, quality=85)

import csv
from pathlib import Path
from collections import defaultdict
import numpy as np
from PIL import Image

def dhash(image, hash_size=8):
    """
    Calculate a simple difference hash (dHash) for an image.
    Resizes the image to (hash_size + 1) x hash_size, converts to grayscale,
    and compares adjacent pixels.
    """
    try:
        # Convert to grayscale and resize
        image = image.convert('L').resize((hash_size + 1, hash_size), Image.Resampling.LANCZOS)
        pixels = np.array(image)
        # Compare adjacent pixels
        diff = pixels[:, 1:] > pixels[:, :-1]
        return diff.flatten()
    except Exception:
        return np.zeros(hash_size * hash_size, dtype=bool)

def hamming_distance(hash1, hash2):
    return np.count_nonzero(hash1 != hash2)

def get_md5(file_path):
    hash_md5 = hashlib.md5()
    with open(file_path, "rb") as f:
        for chunk in iter(lambda: f.read(4096), b""):
            hash_md5.update(chunk)
    return hash_md5.hexdigest()

def get_hashes_from_dir(dir_path):
    hashes = set()
    p = Path(dir_path)
    if not p.exists():
        return hashes
    for ext in ["*.jpg", "*.jpeg", "*.png", "*.webp"]:
        for f in p.rglob(ext):
            hashes.add(get_md5(f))
    return hashes

def get_file_sha256(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        while chunk := f.read(65536):
            h.update(chunk)
    return h.hexdigest()

def main():

    wilt_family_map = {} # orig_name -> family_id
    for idx, f_name in enumerate(wilt_orig_files):
        wilt_family_map[f_name] = f"WILT_FAM_{idx:02d}"

    # Map each of the 287 Wilt images in final dataset to its source family
    # Using pHash distance to find closest orig if it's an augmentation
    import numpy as np
    import cv2
    from PIL import Image

    def compute_phash_val(p):
        img = Image.open(p)
        resized = img.convert('L').resize((32, 32), Image.Resampling.LANCZOS)
        arr = np.array(resized, dtype=np.float32)
        dct = cv2.dct(arr)
        dctlowfreq = dct[:8, :8]
        med = np.median(dctlowfreq[1:, 1:])
        diff = dctlowfreq > med
        packed = np.packbits(diff.flatten())
        return np.frombuffer(packed.tobytes(), dtype='>u8')[0]

    def popcount64(v):
        v = v - ((v >> np.uint64(1)) & np.uint64(0x5555555555555555))
        v = (v & np.uint64(0x3333333333333333)) + ((v >> np.uint64(2)) & np.uint64(0x3333333333333333))
        v = (v + (v >> np.uint64(4))) & np.uint64(0x0f0f0f0f0f0f0f0f)
        return (v * np.uint64(0x0101010101010101)) >> np.uint64(56)

    orig_paths = [wilt_staging_dir / f for f in wilt_orig_files]
    orig_hashes = np.array([compute_phash_val(p) for p in orig_paths], dtype=np.uint64)

    wilt_family_records = []
    wilt_family_members = defaultdict(list)

    for r in wilt_rows:
        img_p = DATASET_ROOT / r["relative_path"]

            "file_sha256": r["file_sha256"]
        }

                    "image_1_class": p["class_1"],
                    "image_1_split": p["split_1"],
                    "image_2_path": p["path_2"],

            "image_2_split", "phash_distance", "dhash_distance", "review_status", "proposed_action", "notes"
        ])
        writer.writeheader()
        writer.writerows(near_review_records)

    # 5. Clean Candidate Pool Assembly
    print("\n[Step 5/5] Building Clean Candidate Pool and Group-Aware Split Manifest...")
    # Safe candidate definition:
    # 1. Not in exact cross-class duplicate list (55 groups / 111 images excluded)

        if inclusion_status == "PROPOSED_CANDIDATE":

    print("\nProposed Clean Pool Counts:")

def parse_ground_truth(gt_path: Path, expected_classes: list):
    """Parse optional ground truth CSV."""
    if not gt_path.exists():
        print(f"ERROR: Ground-truth file not found: {gt_path}")
        sys.exit(1)

    gt_map = {}
    with open(gt_path, mode="r", encoding="utf-8") as f:
        reader = csv.DictReader(f)
        fields = [c.lower().strip() for c in reader.fieldnames]
        
        # Identify image and label columns
        img_col = next((c for c in reader.fieldnames if c.lower().strip() in ["image_name", "filename", "image", "image_path", "file"]), None)
        lbl_col = next((c for c in reader.fieldnames if c.lower().strip() in ["label", "class", "ground_truth", "disease"]), None)

        if not img_col or not lbl_col:
            print(f"ERROR: Could not identify image and label columns in {gt_path}. Found columns: {reader.fieldnames}")
            sys.exit(1)

        for row in reader:
            img_key = Path(row[img_col].strip()).name
            lbl_val = row[lbl_col].strip()
            if lbl_val not in expected_classes:
                print(f"WARNING: Unknown label '{lbl_val}' for image '{img_key}'. Skipping.")
                continue
            gt_map[img_key] = lbl_val

    print(f"Loaded {len(gt_map)} verified ground-truth annotations from {gt_path.name}.")
    return gt_map

def dhash(image, hash_size=8):
    try:
        img = image.convert('L').resize((hash_size + 1, hash_size), Image.Resampling.LANCZOS)
        pixels = np.array(img)
        diff = pixels[:, 1:] > pixels[:, :-1]
        return diff.flatten()
    except Exception:
        return np.zeros(hash_size * hash_size, dtype=bool)

def hamming_distance(h1, h2):
    return np.count_nonzero(h1 != h2)

def create_contact_sheet(image_paths, output_path, cols=10, thumb_size=128):
    if not image_paths:
        return
    rows = (len(image_paths) + cols - 1) // cols
    sheet_width = cols * thumb_size
    sheet_height = rows * thumb_size
    sheet = Image.new("RGB", (sheet_width, sheet_height), color=(255, 255, 255))
    
    for i, img_path in enumerate(image_paths):
        try:
            with Image.open(img_path) as img:
                img_copy = img.copy()
                img_copy.thumbnail((thumb_size, thumb_size))
                x = (i % cols) * thumb_size
                y = (i // cols) * thumb_size
                w, h = img_copy.size
                offset_x = (thumb_size - w) // 2
                offset_y = (thumb_size - h) // 2
                sheet.paste(img_copy, (x + offset_x, y + offset_y))
        except Exception:
            pass
    sheet.save(output_path, quality=85)

def get_file_sha256(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        while chunk := f.read(65536):
            h.update(chunk)
    return h.hexdigest()

def compute_phash_val(p):
    with Image.open(p) as img:
        img_size = 32
        resized = img.convert('L').resize((img_size, img_size), Image.Resampling.LANCZOS)
        arr = np.array(resized, dtype=np.float32)
        dct = cv2.dct(arr)
        dctlowfreq = dct[:8, :8]
        med = np.median(dctlowfreq[1:, 1:])
        diff = dctlowfreq > med
        packed = np.packbits(diff.flatten())
        return np.frombuffer(packed.tobytes(), dtype='>u8')[0]

def popcount64(v):
    v = v - ((v >> np.uint64(1)) & np.uint64(0x5555555555555555))
    v = (v & np.uint64(0x3333333333333333)) + ((v >> np.uint64(2)) & np.uint64(0x3333333333333333))
    v = (v + (v >> np.uint64(4))) & np.uint64(0x0f0f0f0f0f0f0f0f)
    return (v * np.uint64(0x0101010101010101)) >> np.uint64(56)

                 f"Corrupt count: {corrupt_count}, Modes found: {modes}, Formats: {formats}")

        meta_b = json.load(f)
    report_check(7, "Model Checkpoint Byte-Identical Preservation", ckpt_sha == meta_b["checkpoint_sha256"],
                 f"Checkpoint SHA-256: {ckpt_sha[:16]}... (matches pre-build)")

    # 8. Wilt family leakage check

    tt_leak = fams_by_split["train"] & fams_by_split["test"]

    # 10. Crop Isolation check

            non_pigeonpea += 1
    report_check(10, "Strict Crop Isolation Verified", non_pigeonpea == 0,
                 f"Unrelated crop records detected: {non_pigeonpea} / {len(manifest_rows)}")

    print("\n" + "=" * 80)

#!/usr/bin/env python3
"""
TurSathi AI — Field Dataset Collection Tracker and Verification Audit Utility.

Audits field collection manifests for the pigeonpea organ validator without modifying
any source files, images, or application state (read-only audit).

Enforces:
- Required metadata columns and non-null values.
- Allowed class labels and verification statuses.
- Strict isolation of quarantined and rejected images from verified readiness counts.
- Image file existence, decode integrity, and SHA-256 deduplication.
- Zero plant-level and farm-level split leakage across dataset partitions.
- Minimum target thresholds:
    - Tur_Pod: >= 250 verified images
    - Tur_WholePlant_or_Canopy: >= 200 verified images
    - Not_Tur_or_Unsupported: >= 100 verified images
    - Distinct farms: >= 5 geographically separated plots
"""

import argparse
import csv
import hashlib
import json
import logging
import sys
from dataclasses import asdict, dataclass, field
from pathlib import Path
from typing import Any, Dict, List, Optional, Set

try:
    from PIL import Image
    HAS_PIL = True
except ImportError:
    HAS_PIL = False

# ==============================================================================
# Targets and Constraints (from PHASE5D_A_DESIGN_REPORT & FIELD_COLLECTION_PREPARATION)
# ==============================================================================

TARGET_TUR_POD: int = 250
TARGET_TUR_WHOLE_PLANT: int = 200
TARGET_NOT_TUR_DISTRACTOR: int = 100
TARGET_DISTINCT_FARMS: int = 5

REQUIRED_COLUMNS: List[str] = [
    "filename",
    "class_label",
    "plant_id",
    "farm_id",
    "capture_date",
    "verification_status",
    "verifier_reference",
]

ALLOWED_CLASSES: Set[str] = {
    "Tur_Pod",
    "Tur_WholePlant_or_Canopy",
    "Tur_Leaf",
    "Not_Tur_or_Unsupported",
}

ALLOWED_STATUSES: Set[str] = {
    "verified",
    "quarantined",
    "rejected",
}


@dataclass
class AuditIssue:
    severity: str  # "ERROR" or "WARNING"
    category: str
    message: str
    record_index: Optional[int] = None
    filename: Optional[str] = None


@dataclass
class AuditSummary:
    total_records: int = 0
    valid_syntax_records: int = 0
    verified_records: int = 0
    quarantined_records: int = 0
    rejected_records: int = 0
    counts_by_class: Dict[str, Dict[str, int]] = field(default_factory=dict)
    distinct_farms: List[str] = field(default_factory=list)
    distinct_plants_count: int = 0
    verified_by_class: Dict[str, int] = field(default_factory=dict)
    target_checks: Dict[str, Dict[str, Any]] = field(default_factory=dict)
    leakage_detected: bool = False
    is_dataset_ready: bool = False
    blockers: List[str] = field(default_factory=list)
    issues: List[Dict[str, Any]] = field(default_factory=list)


class FieldCollectionAuditor:
    """
    Read-only audit utility for field collection manifests.
    """

    def __init__(
        self,
        manifest_path: Path,
        image_dir: Optional[Path] = None,
        check_images: bool = True,
    ) -> None:
        self.manifest_path = Path(manifest_path)
        self.image_dir = Path(image_dir) if image_dir else None
        self.check_images = check_images
        self.issues: List[AuditIssue] = []

    def log_issue(
        self,
        severity: str,
        category: str,
        message: str,
        record_index: Optional[int] = None,
        filename: Optional[str] = None,
    ) -> None:
        self.issues.append(
            AuditIssue(
                severity=severity,
                category=category,
                message=message,
                record_index=record_index,
                filename=filename,
            )
        )

    def audit(self) -> AuditSummary:
        summary = AuditSummary()

        if not self.manifest_path.exists():
            self.log_issue("ERROR", "MANIFEST_NOT_FOUND", f"Manifest file does not exist: {self.manifest_path}")
            summary.blockers.append(f"Manifest not found: {self.manifest_path}")
            summary.issues = [asdict(i) for i in self.issues]
            return summary

        # 1. Parse CSV
        rows: List[Dict[str, str]] = []
        try:
            with open(self.manifest_path, "r", encoding="utf-8-sig") as f:
                reader = csv.reader(f)
                header: Optional[List[str]] = None
                for line_num, raw_row in enumerate(reader, start=1):
                    # Skip empty rows or comment rows
                    if not raw_row or not any(field.strip() for field in raw_row):
                        continue
                    first_cell = raw_row[0].strip()
                    if first_cell.startswith("#"):
                        continue
                    if header is None:
                        header = [c.strip() for c in raw_row]
                        continue
                    row_dict = {
                        header[i]: raw_row[i].strip() if i < len(raw_row) else ""
                        for i in range(len(header))
                    }
                    rows.append(row_dict)
        except Exception as exc:
            self.log_issue("ERROR", "CSV_PARSE_ERROR", f"Failed to read CSV: {exc}")
            summary.blockers.append(f"CSV parse error: {exc}")
            summary.issues = [asdict(i) for i in self.issues]
            return summary

        summary.total_records = len(rows)

        if not header:
            self.log_issue("ERROR", "EMPTY_MANIFEST", "Manifest does not contain a header line.")
            summary.blockers.append("Manifest header is missing.")
            summary.issues = [asdict(i) for i in self.issues]
            return summary

        # Check required columns
        missing_cols = [c for c in REQUIRED_COLUMNS if c not in header]
        if missing_cols:
            self.log_issue("ERROR", "MISSING_COLUMNS", f"Missing required columns: {missing_cols}")
            summary.blockers.append(f"Missing required columns: {missing_cols}")

        # If zero rows (e.g. empty template), report cleanly without fabricating progress
        if summary.total_records == 0:
            summary.blockers.append("Manifest contains zero collection records (Collection has not started).")
            self._evaluate_readiness(summary, set(), set(), {})
            summary.issues = [asdict(i) for i in self.issues]
            return summary

        # 2. Row-by-Row Validation
        filenames_seen: Set[str] = set()
        file_hashes: Dict[str, str] = {}
        farms_seen: Set[str] = set()
        plants_seen: Set[str] = set()

        # Split leakage tracking: plant_id -> set of splits, farm_id -> set of splits
        plant_to_splits: Dict[str, Set[str]] = {}
        farm_to_splits: Dict[str, Set[str]] = {}

        counts_by_class: Dict[str, Dict[str, int]] = {
            c: {"verified": 0, "quarantined": 0, "rejected": 0, "other": 0}
            for c in ALLOWED_CLASSES
        }
        verified_by_class: Dict[str, int] = {c: 0 for c in ALLOWED_CLASSES}

        valid_records = 0

        for idx, row in enumerate(rows, start=1):
            row_valid = True
            fname = row.get("filename", "")
            clabel = row.get("class_label", "")
            pid = row.get("plant_id", "")
            fid = row.get("farm_id", "")
            status = row.get("verification_status", "").lower()
            split = row.get("split", "").lower()

            # Missing required fields
            for req in REQUIRED_COLUMNS:
                if not row.get(req, ""):
                    self.log_issue("ERROR", "MISSING_VALUE", f"Record #{idx} missing required field '{req}'", idx, fname)
                    row_valid = False

            # Duplicate filename in manifest
            if fname:
                if fname in filenames_seen:
                    self.log_issue("ERROR", "DUPLICATE_FILENAME", f"Duplicate filename '{fname}' at row #{idx}", idx, fname)
                    row_valid = False
                filenames_seen.add(fname)

            # Class label check
            if clabel not in ALLOWED_CLASSES:
                self.log_issue("ERROR", "INVALID_CLASS", f"Unrecognized class '{clabel}' at row #{idx}", idx, fname)
                row_valid = False

            # Verification status check
            if status not in ALLOWED_STATUSES:
                self.log_issue("ERROR", "INVALID_STATUS", f"Unrecognized verification status '{status}' at row #{idx}", idx, fname)
                row_valid = False

            if fid:
                farms_seen.add(fid)
            if pid:
                plants_seen.add(pid)

            # Aggregate raw counts by class (all statuses, regardless of row_valid)
            if clabel in counts_by_class:
                if status in counts_by_class[clabel]:
                    counts_by_class[clabel][status] += 1
                else:
                    counts_by_class[clabel]["other"] += 1

            # Count non-verified statuses immediately (quarantine/reject are informational)
            if status == "quarantined":
                summary.quarantined_records += 1
            elif status == "rejected":
                summary.rejected_records += 1
            # NOTE: verified_records and verified_by_class are deferred to after all
            # row-level checks complete so that image-existence and decode failures
            # cannot inflate the verified count (see fix applied at end of row loop).

            # Split leakage tracking if split column is provided
            if split:
                if pid:
                    plant_to_splits.setdefault(pid, set()).add(split)
                if fid:
                    farm_to_splits.setdefault(fid, set()).add(split)

            # Image existence and decodability check
            if self.image_dir and fname:
                img_path = self.image_dir / fname
                if not img_path.exists():
                    self.log_issue("ERROR", "FILE_NOT_FOUND", f"Image file '{fname}' not found in {self.image_dir}", idx, fname)
                    row_valid = False
                elif self.check_images and HAS_PIL:
                    try:
                        with Image.open(img_path) as img:
                            img.verify()
                        with Image.open(img_path) as img:
                            w, h = img.size
                            if w < 224 or h < 224:
                                self.log_issue("WARNING", "IMAGE_BELOW_MIN_DIMENSION", f"Image '{fname}' resolution {w}x{h} < 224x224", idx, fname)
                    except Exception as img_exc:
                        self.log_issue("ERROR", "CORRUPT_IMAGE", f"Image '{fname}' cannot be decoded: {img_exc}", idx, fname)
                        row_valid = False

                # Hash deduplication check
                if img_path.exists():
                    try:
                        h_sha = hashlib.sha256()
                        with open(img_path, "rb") as f_img:
                            while chunk := f_img.read(65536):
                                h_sha.update(chunk)
                        sha_digest = h_sha.hexdigest()
                        if sha_digest in file_hashes:
                            self.log_issue(
                                "ERROR",
                                "EXACT_DUPLICATE_HASH",
                                f"Image '{fname}' has identical SHA-256 hash to '{file_hashes[sha_digest]}'",
                                idx,
                                fname,
                            )
                            row_valid = False
                        else:
                            file_hashes[sha_digest] = fname
                    except Exception as h_exc:
                        self.log_issue("WARNING", "HASH_ERROR", f"Failed to hash '{fname}': {h_exc}", idx, fname)

            if row_valid:
                valid_records += 1
                # Fix 1: count toward verified only when the entire row is valid
                # (schema, file existence, decode integrity, and no duplicate hash).
                if status == "verified" and clabel in verified_by_class:
                    verified_by_class[clabel] += 1
                    summary.verified_records += 1

        summary.valid_syntax_records = valid_records
        summary.counts_by_class = counts_by_class
        summary.verified_by_class = verified_by_class
        summary.distinct_farms = sorted(list(farms_seen))
        summary.distinct_plants_count = len(plants_seen)

        # 3. Check for Split Leakage
        leakage_detected = False
        for pid, splits in plant_to_splits.items():
            if len(splits) > 1:
                leakage_detected = True
                self.log_issue(
                    "ERROR",
                    "PLANT_SPLIT_LEAKAGE",
                    f"Plant ID '{pid}' appears in multiple splits: {sorted(list(splits))}",
                )
                summary.blockers.append(f"Plant ID '{pid}' leaked across splits: {sorted(list(splits))}")

        for fid, splits in farm_to_splits.items():
            if len(splits) > 1:
                leakage_detected = True
                self.log_issue(
                    "ERROR",
                    "FARM_SPLIT_LEAKAGE",
                    f"Farm ID '{fid}' appears in multiple splits: {sorted(list(splits))}",
                )
                summary.blockers.append(f"Farm ID '{fid}' leaked across splits: {sorted(list(splits))}")

        summary.leakage_detected = leakage_detected

        # 4. Target Threshold Evaluations
        self._evaluate_readiness(summary, farms_seen, plants_seen, verified_by_class)

        summary.issues = [asdict(i) for i in self.issues]
        return summary

    def _evaluate_readiness(
        self,
        summary: AuditSummary,
        farms_seen: Set[str],
        plants_seen: Set[str],
        verified_by_class: Dict[str, int],
    ) -> None:
        pod_verified = verified_by_class.get("Tur_Pod", 0)
        canopy_verified = verified_by_class.get("Tur_WholePlant_or_Canopy", 0)
        distractor_verified = verified_by_class.get("Not_Tur_or_Unsupported", 0)
        farm_count = len(farms_seen)

        summary.target_checks = {
            "Tur_Pod": {
                "target": TARGET_TUR_POD,
                "verified": pod_verified,
                "satisfied": pod_verified >= TARGET_TUR_POD,
                "deficit": max(0, TARGET_TUR_POD - pod_verified),
            },
            "Tur_WholePlant_or_Canopy": {
                "target": TARGET_TUR_WHOLE_PLANT,
                "verified": canopy_verified,
                "satisfied": canopy_verified >= TARGET_TUR_WHOLE_PLANT,
                "deficit": max(0, TARGET_TUR_WHOLE_PLANT - canopy_verified),
            },
            "Not_Tur_or_Unsupported": {
                "target": TARGET_NOT_TUR_DISTRACTOR,
                "verified": distractor_verified,
                "satisfied": distractor_verified >= TARGET_NOT_TUR_DISTRACTOR,
                "deficit": max(0, TARGET_NOT_TUR_DISTRACTOR - distractor_verified),
            },
            "Distinct_Farms": {
                "target": TARGET_DISTINCT_FARMS,
                "actual": farm_count,
                "satisfied": farm_count >= TARGET_DISTINCT_FARMS,
                "deficit": max(0, TARGET_DISTINCT_FARMS - farm_count),
            },
        }

        # Determine readiness blockers
        if pod_verified < TARGET_TUR_POD:
            summary.blockers.append(f"Tur_Pod deficit: {pod_verified}/{TARGET_TUR_POD} verified images.")
        if canopy_verified < TARGET_TUR_WHOLE_PLANT:
            summary.blockers.append(f"Tur_WholePlant_or_Canopy deficit: {canopy_verified}/{TARGET_TUR_WHOLE_PLANT} verified images.")
        if distractor_verified < TARGET_NOT_TUR_DISTRACTOR:
            summary.blockers.append(f"Not_Tur_or_Unsupported deficit: {distractor_verified}/{TARGET_NOT_TUR_DISTRACTOR} verified images.")
        if farm_count < TARGET_DISTINCT_FARMS:
            summary.blockers.append(f"Distinct farm deficit: {farm_count}/{TARGET_DISTINCT_FARMS} distinct farms.")
        if summary.quarantined_records > 0:
            summary.blockers.append(f"Unresolved quarantine: {summary.quarantined_records} records require agronomist review.")
        if summary.leakage_detected:
            summary.blockers.append("Split leakage detected: Plant or farm IDs repeated across dataset partitions.")

        error_issues = [i for i in self.issues if i.severity == "ERROR"]
        if error_issues:
            summary.blockers.append(f"Manifest contains {len(error_issues)} error issues requiring resolution.")

        # Ready ONLY if all targets met, zero leakages, zero critical errors, zero unreviewed quarantine
        summary.is_dataset_ready = len(summary.blockers) == 0


def format_audit_report(summary: AuditSummary) -> str:
    lines = []
    lines.append("=" * 70)
    lines.append("TUR SATHI AI — FIELD DATASET COLLECTION AUDIT REPORT")
    lines.append("=" * 70)
    lines.append(f"Total Collection Records:     {summary.total_records}")
    lines.append(f"Verified Records:             {summary.verified_records}")
    lines.append(f"Quarantined Records:          {summary.quarantined_records}")
    lines.append(f"Rejected Records:             {summary.rejected_records}")
    lines.append(f"Distinct Farms:               {len(summary.distinct_farms)} -> {summary.distinct_farms}")
    lines.append(f"Distinct Plants:              {summary.distinct_plants_count}")
    lines.append("-" * 70)
    lines.append("PROGRESS VS TARGET THRESHOLDS:")
    for k, v in summary.target_checks.items():
        curr = v.get("verified", v.get("actual", 0))
        tgt = v["target"]
        sat = "PASS [OK]" if v["satisfied"] else f"DEFICIT (-{v['deficit']})"
        lines.append(f"  {k:<26}: {curr:>4} / {tgt:>4}  [{sat}]")
    lines.append("-" * 70)
    lines.append(f"DATASET READY FOR TRAINING:   {'YES [APPROVED]' if summary.is_dataset_ready else 'NO [HOLD]'}")
    if summary.blockers:
        lines.append("\nACTIVE BLOCKERS:")
        for b in summary.blockers:
            lines.append(f"  * {b}")
    lines.append("=" * 70)
    return "\n".join(lines)

    sys.exit(0 if summary.is_dataset_ready else 1)


if __name__ == "__main__":
    main()

import csv
import hashlib
from pathlib import Path
from collections import defaultdict

def get_md5(file_path):
    hash_md5 = hashlib.md5()
    with open(file_path, "rb") as f:
        for chunk in iter(lambda: f.read(4096), b""):
            hash_md5.update(chunk)
    return hash_md5.hexdigest()

def get_hashes_from_dir(dir_path_str):
    hashes = set()
    p = Path(dir_path_str)
    if not p.exists():
        return hashes
    for ext in ["*.jpg", "*.jpeg", "*.png", "*.webp"]:
        for f in p.rglob(ext):
            hashes.add(get_md5(f))
    return hashes

    if dir_b.exists():
        for ext in ["*.jpg", "*.jpeg", "*.png", "*.webp"]:
            for f in dir_b.rglob(ext):
                if get_md5(f) not in v2_hashes:
                    candidate_files.append(f)
                    
    total_candidates = len(candidate_files)
    
    # Recalculate duplicate groups
    hash_to_files = defaultdict(list)
    for f in candidate_files:
        md5 = get_md5(f)
        hash_to_files[md5].append(f)
        
    unique_hashes = len(hash_to_files)
    
    # Calculate exact holdout duplicates
    holdout_dups_count = 0
    holdout_dup_files = []
    
    # Calculate exact internal duplicates
    # Internal dup files = sum(len(files) - 1 for files in hash_to_files if len(files) > 1)
    # BUT wait! If an image is a holdout dup AND has internal copies, how was it counted in 4C/4D?
    # This is likely the source of the 2-image discrepancy.
    internal_dups_count = 0
    internal_dup_files = []
    
    clean_candidates_count = 0
    
    for md5, files in hash_to_files.items():
        is_holdout = md5 in holdout_hashes
        
        # Determine the primary (first seen) file vs duplicates
        primary = files[0]
        copies = files[1:]
        
        if is_holdout:
            # ALL files with this hash are holdout duplicates (they cannot be clean candidates)
            holdout_dups_count += len(files)
            holdout_dup_files.extend(files)
            # Should they also be counted as internal duplicates if len(files) > 1?
            # If so, they are double counted in 4C/4D! 
            # In an exclusive categorization:
            # A file is EITHER holdout dup OR internal dup OR clean.
            # If we prioritize HOLDOUT, then internal_dups = 0 for this group.
            pass
        else:
            # If not a holdout dup, the primary is clean, copies are internal dups.
            clean_candidates_count += 1
            internal_dups_count += len(copies)
            internal_dup_files.extend(copies)

    manifest_rows = []

                elif "pigeonpea_disease_final" in f.parts:

    code("""
# 1. Compute 1280-D Centroids from Training Set
model.eval()
class_features = {c: [] for c in range(len(EXPECTED_CLASSES))}
with torch.no_grad():
    for inputs, tgts in train_loader:
        inputs = inputs.to(DEVICE)
        feats = model.extract_features(inputs).cpu().numpy()
        for f, t in zip(feats, tgts.numpy()):
            class_features[t].append(f)

centroids = {}
norm_centroids = {}
for c in range(len(EXPECTED_CLASSES)):
    c_feats = np.array(class_features[c])
    cent = np.mean(c_feats, axis=0)
    centroids[EXPECTED_CLASSES[c]] = cent
    norm_centroids[c] = cent / (np.linalg.norm(cent) + 1e-8)

# 2. Calibrate Per-Class Cosine Thresholds on Validation Set
val_class_dists = {c: [] for c in range(len(EXPECTED_CLASSES))}
with torch.no_grad():
    for inputs, tgts in val_loader:
        inputs = inputs.to(DEVICE)
        feats = model.extract_features(inputs).cpu().numpy()
        for f, t in zip(feats, tgts.numpy()):
            norm_v = f / (np.linalg.norm(f) + 1e-8)
            d = float(1.0 - np.dot(norm_v, norm_centroids[t]))
            val_class_dists[t].append(d)

per_class_thresholds = {}
for c in range(len(EXPECTED_CLASSES)):
    c_name = EXPECTED_CLASSES[c]
    p975 = float(np.percentile(val_class_dists[c], 97.5))
    if c_name == "Wilt":
        p975 = min(p975, 0.40) # Strict ceiling preventing Wilt Sinkhole
    per_class_thresholds[c_name] = round(p975, 4)

print("Per-Class Calibrated Cosine Distance Thresholds:")
for c_name, th in per_class_thresholds.items():
    print(f"  - {c_name:18s}: {th:.4f}")

# 3. Test Against External OOD Evaluation Images
OOD_DIR = DATA_DIR / "ood_evaluation"
ood_files = [f for f in OOD_DIR.iterdir() if f.is_file() and f.suffix.lower() in ['.jpg','.jpeg','.png','.webp']] if OOD_DIR.exists() else []

rejected = 0
wilt_sinkhole_count = 0
for f in ood_files:
    img = Image.open(f).convert("RGB")
    tensor = eval_transform(img).unsqueeze(0).to(DEVICE)
    with torch.no_grad():
        logits, feat = model(tensor)
        probs = torch.softmax(logits, dim=1)[0].cpu().numpy()
    pred_idx = np.argmax(probs)
    pred_class = EXPECTED_CLASSES[pred_idx]
    conf = float(probs[pred_idx])
    sorted_p = np.sort(probs)[::-1]
    margin = float(sorted_p[0] - sorted_p[1]) if len(sorted_p) > 1 else 1.0

    feat_v = feat[0].cpu().numpy()
    norm_v = feat_v / (np.linalg.norm(feat_v) + 1e-8)
    cos_dist = float(1.0 - np.dot(norm_v, norm_centroids[pred_idx]))

    th = per_class_thresholds.get(pred_class, 0.45)
    is_rejected = (cos_dist > th) or (conf < 0.70) or (margin < 0.20)
    if is_rejected:
        rejected += 1
    else:
        if pred_class == "Wilt":
            wilt_sinkhole_count += 1

rej_rate = (rejected / len(ood_files) * 100) if ood_files else 0
print(f"\\nOOD Rejection Rate on External Unseen Images: {rejected}/{len(ood_files)} ({rej_rate:.1f}%)")

""")

#!/usr/bin/env python3
"""
TurSathi AI - Dataset Integrity Analysis, Conflict Quarantine & Clean Split Generator
Phase B Implementation:
1. Cryptographically hashes (SHA-256 and MD5) all images in datasets/pigeonpea_disease_final/.
2. Identifies all cross-class duplicate conflicts and cross-split duplicates.
3. Generates a structured duplicate/conflict audit CSV and JSON report in reports/.
4. Constructs a clean, leak-free, conflict-free stratified split manifest (70% train / 15% val / 15% test).
5. Quarantines ambiguous conflicting image hashes without deleting or moving any source files.
"""

import os
import csv
import json
import hashlib
import random
from pathlib import Path
from collections import defaultdict, Counter

PROJECT_ROOT = Path(__file__).resolve().parent.parent
SOURCE_DATASET_DIR = PROJECT_ROOT / "datasets" / "pigeonpea_disease_final"
REPORTS_DIR = PROJECT_ROOT / "reports"
RETRAINING_DIR = PROJECT_ROOT / "ml" / "retraining"

REPORTS_DIR.mkdir(parents=True, exist_ok=True)
RETRAINING_DIR.mkdir(parents=True, exist_ok=True)

SEED = 42
VALID_EXTENSIONS = {".jpg", ".jpeg", ".png", ".webp", ".bmp"}
EXPECTED_CLASSES = ["Healthy", "Leaf_Spot", "Leaf_webber", "Sterilic_mosaic", "Wilt"]

def run_dataset_audit():
    print("=" * 70)
    print("TUR SATHI AI - DATASET INTEGRITY & CONFLICT AUDIT")
    print("=" * 70)

    # 1. Read all image records
    records = []
    splits = ["train", "val", "test"]

    for split in splits:
        split_dir = SOURCE_DATASET_DIR / split
        if not split_dir.exists():
            continue
        for class_name in sorted(os.listdir(split_dir)):
            c_dir = split_dir / class_name
            if not c_dir.is_dir():
                continue
            for fname in sorted(os.listdir(c_dir)):
                fp = c_dir / fname
                if not fp.is_file() or fname.startswith("."):
                    continue
                if fp.suffix.lower() not in VALID_EXTENSIONS:
                    continue
                
                file_bytes = fp.read_bytes()
                sha256 = hashlib.sha256(file_bytes).hexdigest()
                md5 = hashlib.md5(file_bytes).hexdigest()
                size = fp.stat().st_size
                rel_path = fp.relative_to(PROJECT_ROOT).as_posix()

                records.append({
                    "split": split,
                    "class": class_name,
                    "filename": fname,
                    "relative_path": rel_path,
                    "size_bytes": size,
                    "md5": md5,
                    "sha256": sha256
                })

    total_records = len(records)
    print(f"Total image files discovered in production dataset: {total_records}")

    # 2. Group by SHA-256
    by_hash = defaultdict(list)
    for r in records:
        by_hash[r["sha256"]].append(r)

    total_unique_hashes = len(by_hash)
    print(f"Total unique cryptographic SHA-256 hashes: {total_unique_hashes}")

    # 3. Detect duplicate groups and label conflicts
    conflict_groups = []
    same_class_dup_groups = []

    group_id = 1
    for h, group in by_hash.items():
        if len(group) > 1:
            classes = sorted(set(x["class"] for x in group))
            splits_present = sorted(set(x["split"] for x in group))
            if len(classes) > 1:
                conflict_groups.append({
                    "group_id": group_id,
                    "sha256": h,
                    "classes": classes,
                    "splits": splits_present,
                    "count": len(group),
                    "items": group
                })
                group_id += 1
            else:
                same_class_dup_groups.append({
                    "group_id": group_id,
                    "sha256": h,
                    "class": classes[0],
                    "splits": splits_present,
                    "count": len(group),
                    "items": group
                })
                group_id += 1

    print(f"Cross-class label conflict groups: {len(conflict_groups)}")
    print(f"Same-class cross-split duplicate groups: {len(same_class_dup_groups)}")

    # 4. Generate duplicate and conflict CSV report
    csv_path = REPORTS_DIR / "duplicate_and_label_conflicts.csv"
    with open(csv_path, "w", newline="", encoding="utf-8") as f:
        writer = csv.writer(f)
        writer.writerow([
            "Group_ID", "Conflict_Type", "SHA256", "Contradictory_Classes",
            "Splits_Involved", "File_Count", "File_Path", "Original_Split", "Original_Class"
        ])
        for grp in conflict_groups:
            c_str = " | ".join(grp["classes"])
            s_str = " | ".join(grp["splits"])
            for it in grp["items"]:
                writer.writerow([
                    grp["group_id"], "CROSS_CLASS_LABEL_CONFLICT", grp["sha256"],
                    c_str, s_str, grp["count"], it["relative_path"], it["split"], it["class"]
                ])
        for grp in same_class_dup_groups:
            s_str = " | ".join(grp["splits"])
            for it in grp["items"]:
                writer.writerow([
                    grp["group_id"], "SAME_CLASS_DUPLICATE", grp["sha256"],
                    grp["class"], s_str, grp["count"], it["relative_path"], it["split"], it["class"]
                ])
    print(f"Wrote detailed duplicate review CSV -> {csv_path.relative_to(PROJECT_ROOT)}")

    # 5. Build Clean Stratified Split Manifest
    # Quarantine any hash in conflict_groups
    quarantined_hashes = set(x["sha256"] for x in conflict_groups)
    print(f"Total quarantined image hashes (due to conflicting labels): {len(quarantined_hashes)}")

    # Gather clean unique items
    clean_items_by_class = defaultdict(list)
    manifest_records = []

    # First, record quarantined files
    for grp in conflict_groups:
        for it in grp["items"]:
            manifest_records.append({
                "sha256": it["sha256"],
                "md5": it["md5"],
                "source_relative_path": it["relative_path"],
                "filename": it["filename"],
                "original_split": it["split"],
                "original_class": it["class"],
                "clean_split": "quarantine",
                "clean_class": "quarantine",
                "status": "quarantined_conflict",
                "quarantine_reason": f"Conflicting labels: {', '.join(grp['classes'])} (Group {grp['group_id']})"
            })

    # For valid hashes, pick ONE representative file per unique hash
    valid_hashes = set(by_hash.keys()) - quarantined_hashes

    # Track same-class duplicates (keep 1st, mark others as deduplicated)
    for h in sorted(valid_hashes):
        group = by_hash[h]
        rep = group[0]
        clean_items_by_class[rep["class"]].append(rep)
        if len(group) > 1:
            for extra in group[1:]:
                manifest_records.append({
                    "sha256": extra["sha256"],
                    "md5": extra["md5"],
                    "source_relative_path": extra["relative_path"],
                    "filename": extra["filename"],
                    "original_split": extra["split"],
                    "original_class": extra["class"],
                    "clean_split": "quarantine",
                    "clean_class": extra["class"],
                    "status": "quarantined_deduplicate",
                    "quarantine_reason": f"Exact duplicate of representative {rep['relative_path']}"
                })

    # Stratified split for clean unique images: 70% train, 15% val, 15% test
    random.seed(SEED)
    clean_split_counts = defaultdict(lambda: Counter())

    for c in EXPECTED_CLASSES:
        items = clean_items_by_class[c]
        random.shuffle(items)
        n = len(items)
        n_train = int(round(n * 0.70))
        n_val = int(round(n * 0.15))
        # Ensure remaining go to test
        n_test = n - n_train - n_val

        train_set = items[:n_train]
        val_set = items[n_train:n_train + n_val]
        test_set = items[n_train + n_val:]

        for it in train_set:
            clean_split_counts[c]["train"] += 1
            manifest_records.append({
                "sha256": it["sha256"],
                "md5": it["md5"],
                "source_relative_path": it["relative_path"],
                "filename": it["filename"],
                "original_split": it["split"],
                "original_class": it["class"],
                "clean_split": "train",
                "clean_class": c,
                "status": "included",
                "quarantine_reason": None
            })

        for it in val_set:
            clean_split_counts[c]["val"] += 1
            manifest_records.append({
                "sha256": it["sha256"],
                "md5": it["md5"],
                "source_relative_path": it["relative_path"],
                "filename": it["filename"],
                "original_split": it["split"],
                "original_class": it["class"],
                "clean_split": "val",
                "clean_class": c,
                "status": "included",
                "quarantine_reason": None
            })

        for it in test_set:
            clean_split_counts[c]["test"] += 1
            manifest_records.append({
                "sha256": it["sha256"],
                "md5": it["md5"],
                "source_relative_path": it["relative_path"],
                "filename": it["filename"],
                "original_split": it["split"],
                "original_class": it["class"],
                "clean_split": "test",
                "clean_class": c,
                "status": "included",
                "quarantine_reason": None
            })

    print("\nClean Dataset Stratified Split (Leak-Free & Conflict-Free):")
    print(f"{'Class':18s} | {'Train':>6s} | {'Val':>6s} | {'Test':>6s} | {'Total Clean':>11s}")
    print("-" * 55)
    total_train = total_val = total_test = 0
    for c in EXPECTED_CLASSES:
        tr = clean_split_counts[c]["train"]
        va = clean_split_counts[c]["val"]
        te = clean_split_counts[c]["test"]
        tot = tr + va + te
        total_train += tr
        total_val += va
        total_test += te
        print(f"{c:18s} | {tr:6d} | {va:6d} | {te:6d} | {tot:11d}")
    print("-" * 55)
    total_clean = total_train + total_val + total_test
    print(f"{'TOTAL':18s} | {total_train:6d} | {total_val:6d} | {total_test:6d} | {total_clean:11d}")
    print(f"Quarantined files: {total_records - total_clean} ({len(conflict_groups)} unique conflict hashes)")

    # 6. Save Clean Manifest (JSON and CSV)
    clean_manifest_json = RETRAINING_DIR / "clean_dataset_manifest.json"
    clean_manifest_csv = RETRAINING_DIR / "clean_dataset_manifest.csv"

    manifest_data = {
        "dataset_name": "TurSathi Pigeonpea Clean Retraining Dataset",
        "created_at": "2026-10-26",
        "random_seed": SEED,
        "classes": EXPECTED_CLASSES,
        "summary": {
            "total_source_files": total_records,
            "total_unique_hashes": total_unique_hashes,
            "total_quarantined_files": total_records - total_clean,
            "total_clean_unique_images": total_clean,
            "splits": {
                "train": total_train,
                "val": total_val,
                "test": total_test
            },
            "per_class_clean_counts": {c: dict(clean_split_counts[c]) for c in EXPECTED_CLASSES}
        },
        "records": manifest_records
    }

    with open(clean_manifest_json, "w", encoding="utf-8") as f:
        json.dump(manifest_data, f, indent=2)
    print(f"\nWrote Clean Manifest JSON -> {clean_manifest_json.relative_to(PROJECT_ROOT)}")

    with open(clean_manifest_csv, "w", newline="", encoding="utf-8") as f:
        writer = csv.DictWriter(f, fieldnames=[
            "sha256", "md5", "source_relative_path", "filename", "original_split",
            "original_class", "clean_split", "clean_class", "status", "quarantine_reason"
        ])
        writer.writeheader()
        writer.writerows(manifest_records)
    print(f"Wrote Clean Manifest CSV -> {clean_manifest_csv.relative_to(PROJECT_ROOT)}")

    # 7. Save Comprehensive Dataset Integrity Report
    integrity_report_json = REPORTS_DIR / "dataset_integrity_report.json"
    report_content = {
        "audit_timestamp": "2026-10-26",
        "source_dataset": str(SOURCE_DATASET_DIR.relative_to(PROJECT_ROOT)),
        "classes": EXPECTED_CLASSES,
        "total_files_audited": total_records,
        "unique_hashes": total_unique_hashes,
        "duplicate_groups_count": len(conflict_groups) + len(same_class_dup_groups),
        "conflicting_label_groups_count": len(conflict_groups),
        "conflicting_files_count": sum(grp["count"] for grp in conflict_groups),
        "conflict_combinations": dict(Counter(" & ".join(grp["classes"]) for grp in conflict_groups)),
        "quarantined_count": total_records - total_clean,
        "clean_dataset_size": total_clean,
        "clean_split_summary": {
            "train": total_train,
            "val": total_val,
            "test": total_test,
            "by_class": {c: dict(clean_split_counts[c]) for c in EXPECTED_CLASSES}
        }
    }

    with open(integrity_report_json, "w", encoding="utf-8") as f:
        json.dump(report_content, f, indent=2)
    print(f"Wrote Dataset Integrity Report JSON -> {integrity_report_json.relative_to(PROJECT_ROOT)}")
    print("=" * 70)

if __name__ == "__main__":
    run_dataset_audit()

#!/usr/bin/env python3
"""
TurSathi AI - Package Clean Retraining Dataset for Google Colab
Phase C Implementation:
Reads ml/retraining/clean_dataset_manifest.json and packages strictly the clean,
non-quarantined images directly into a single zip file for upload to Google Colab.
Includes the 41 external OOD evaluation images for real-world negative rejection testing.
"""

import os
import json
import zipfile
from pathlib import Path

PROJECT_ROOT = Path(__file__).resolve().parent.parent
MANIFEST_FILE = PROJECT_ROOT / "ml" / "retraining" / "clean_dataset_manifest.json"
OOD_SOURCE_DIR = PROJECT_ROOT / "datasets" / "new_disease_images"
OUTPUT_ZIP = PROJECT_ROOT / "ml" / "retraining" / "TurSathi_Pigeonpea_Clean_Dataset.zip"

VALID_EXTENSIONS = {".jpg", ".jpeg", ".png", ".webp", ".bmp"}

def build_clean_zip():
    print("=" * 70)
    print("TUR SATHI AI - PACKAGING CLEAN RETRAINING DATASET")
    print("=" * 70)

    if not MANIFEST_FILE.exists():
        raise FileNotFoundError(f"Clean manifest not found at: {MANIFEST_FILE}")

    with open(MANIFEST_FILE, "r", encoding="utf-8") as f:
        manifest = json.load(f)

    records = manifest.get("records", [])
    included_records = [r for r in records if r.get("status") == "included"]
    print(f"Total included clean images to package: {len(included_records)}")

    # Check disk space
    import shutil
    _, _, free_bytes = shutil.disk_usage(PROJECT_ROOT)
    free_mb = free_bytes / (1024 * 1024)
    print(f"Available disk space: {free_mb:.1f} MB (Expected zip size: ~85 MB)")

    packaged_count = 0
    with zipfile.ZipFile(OUTPUT_ZIP, "w", zipfile.ZIP_DEFLATED) as zip_out:
        # 1. Write clean split files: train/<class>/..., val/<class>/..., test/<class>/...
        for rec in included_records:
            src_path = PROJECT_ROOT / rec["source_relative_path"]
            if not src_path.exists():
                print(f"Warning: source file missing: {src_path}")
                continue

            # Arcname inside zip: <clean_split>/<clean_class>/<filename>
            arcname = f"{rec['clean_split']}/{rec['clean_class']}/{rec['filename']}"
            zip_out.write(src_path, arcname)
            packaged_count += 1

        # 2. Write external OOD evaluation images
        ood_count = 0
        if OOD_SOURCE_DIR.exists():
            for f in sorted(os.listdir(OOD_SOURCE_DIR)):
                fp = OOD_SOURCE_DIR / f
                if fp.is_file() and fp.suffix.lower() in VALID_EXTENSIONS:
                    arcname = f"ood_evaluation/{f}"
                    zip_out.write(fp, arcname)
                    ood_count += 1
        print(f"Packaged {ood_count} external OOD evaluation images.")

        # 3. Add manifest metadata to zip
        zip_out.write(MANIFEST_FILE, "clean_dataset_manifest.json")

    size_mb = OUTPUT_ZIP.stat().st_size / (1024 * 1024)
    print(f"\nSuccessfully packaged clean dataset into:\n  {OUTPUT_ZIP.resolve()}")
    print(f"Total files in archive: {packaged_count} in-distribution + {ood_count} OOD images")
    print(f"Archive Size: {size_mb:.2f} MB")
    print("=" * 70)

if __name__ == "__main__":
    build_clean_zip()

#!/usr/bin/env python3
"""
TurSathi AI - Dataset Packaging for Google Colab
Packages the 5-class pigeon pea dataset (train, val, test) + OOD evaluation images
into a single, clean zip archive ready for Colab upload.
"""

import os
import zipfile
from pathlib import Path

PROJECT_ROOT = Path(__file__).resolve().parent.parent
SOURCE_DATASET_DIR = PROJECT_ROOT / "datasets" / "pigeonpea_disease_final"
OOD_SOURCE_DIR = PROJECT_ROOT / "datasets" / "new_disease_images"
OUTPUT_ZIP = PROJECT_ROOT / "Tur_Disease_Dataset_Colab_Ready.zip"

VALID_EXTENSIONS = {".jpg", ".jpeg", ".png", ".webp", ".bmp"}

def build_zip():
    print(f"Packaging dataset into: {OUTPUT_ZIP}...")
    file_count = 0

    with zipfile.ZipFile(OUTPUT_ZIP, "w", zipfile.ZIP_DEFLATED) as zip_out:
        # 1. Package 5-class in-distribution splits (train, val, test)
        for split in ["train", "val", "test"]:
            split_dir = SOURCE_DATASET_DIR / split
            if not split_dir.exists():
                print(f"Warning: {split_dir} does not exist.")
                continue
            for root, _, files in os.walk(split_dir):
                for f in sorted(files):
                    f_path = Path(root) / f
                    if f_path.suffix.lower() in VALID_EXTENSIONS:
                        # e.g., train/Healthy/image.jpg
                        arcname = f_path.relative_to(SOURCE_DATASET_DIR).as_posix()
                        zip_out.write(f_path, arcname)
                        file_count += 1

        # 2. Package OOD external evaluation images
        if OOD_SOURCE_DIR.exists():
            for f in sorted(os.listdir(OOD_SOURCE_DIR)):
                f_path = OOD_SOURCE_DIR / f
                if f_path.is_file() and f_path.suffix.lower() in VALID_EXTENSIONS:
                    arcname = f"ood_evaluation/{f}"
                    zip_out.write(f_path, arcname)
                    file_count += 1

    size_mb = OUTPUT_ZIP.stat().st_size / (1024 * 1024)
    print(f"Successfully packaged {file_count} images into {OUTPUT_ZIP.name} ({size_mb:.2f} MB)")

if __name__ == "__main__":
    build_zip()

def sha256_file(path):
    h = hashlib.sha256()
    with open(str(path), "rb") as f:
        for chunk in iter(lambda: f.read(65536), b""):
            h.update(chunk)
    return h.hexdigest()


def is_valid_image(path):
    try:
        with Image.open(str(path)) as img:
            img.verify()
        with Image.open(str(path)) as img:
            img.load()
        return True
    except Exception:
        return False


def collect_from_dirs(dirs):
    imgs = []
    for d in dirs:
        d = Path(d)
        if not d.exists():
            continue
        for f in sorted(d.iterdir()):
            if f.is_file() and f.suffix.lower() in VALID_EXT:
                imgs.append(f)
    return imgs


def dedup(images):
    seen, unique, dupes = set(), [], 0
    for img in images:
        h = sha256_file(img)
        if h not in seen:
            seen.add(h); unique.append(img)
        else:
            dupes += 1
    print(f"    Dedup: {len(images)} -> {len(unique)} unique ({dupes} removed)")
    return unique


def verify(images):
    valid, bad = [], []
    for img in images:
        (valid if is_valid_image(img) else bad).append(img)
    if bad:
        print(f"    WARNING: {len(bad)} corrupted images skipped")
    return valid, bad


def copy_to(imgs, out_dir, prefix, split_name):
    out_dir.mkdir(parents=True, exist_ok=True)
    for i, src in enumerate(imgs):
        ext = Path(src).suffix.lower()
        shutil.copy2(str(src), str(out_dir / f"{prefix}_{split_name}_{i:04d}{ext}"))

            n_train, n_val, n_test = TARGET_TRAIN, TARGET_VAL, TARGET_TEST
            print(f"  OK: Using {usable} of {len(unique)} available.")

        # Ensure augmented families stay together to prevent data leakage across splits
        groups = {}
        for img in unique:
            name = img.stem.lower()
            # Strip common augmentation suffixes
            for kw in ['_aug', '_flip', '_rot', '_bright', '_noise', '_blur', '_crop']:
                if kw in name:
                    name = name[:name.find(kw)]
            groups.setdefault(name, []).append(img)
            
        group_keys = list(groups.keys())
        random.seed(RANDOM_SEED)
        random.shuffle(group_keys)
        
        train_imgs, val_imgs, test_imgs = [], [], []
        # Distribute whole families instead of individual images
        for key in group_keys:
            family = groups[key]
            if len(train_imgs) < n_train:
                train_imgs.extend(family)
            elif len(val_imgs) < n_val:
                val_imgs.extend(family)
            else:
                test_imgs.extend(family)
        
        # Truncate to exact target sizes. Since a family is only added to one list, 
        # dropping the excess prevents leakage across splits.
        train_imgs = train_imgs[:n_train]
        val_imgs = val_imgs[:n_val]
        test_imgs = test_imgs[:n_test]
        print(f"  [4] Split: train={len(train_imgs)}, val={len(val_imgs)}, test={len(test_imgs)}")

        # Clear & rebuild output
        prefix = cls.lower()
        out = {sp: OUTPUT_DIR / sp / cls for sp in SPLITS}
        for d in out.values():
            if d.exists(): shutil.rmtree(d)
            d.mkdir(parents=True)

        copy_to(train_imgs, out["train"], prefix, "train")
        copy_to(val_imgs,   out["val"],   prefix, "val")
        copy_to(test_imgs,  out["test"],  prefix, "test")

        a = {sp: len(list(out[sp].iterdir())) for sp in SPLITS}
        a["total"] = sum(a.values())
        class_results[cls] = a
        print(f"  DONE: train={a['train']}, val={a['val']}, test={a['test']}, total={a['total']}")
        print()

    # Cleanup staging
    if STAGING_DIR.exists():
        shutil.rmtree(STAGING_DIR)
    print("Staging temp cleaned up.")
    print()

    # ── Cross-split leakage check ─────────────────────────────────────────────
    print("=" * 70)
    print("CROSS-SPLIT LEAKAGE CHECK (SHA256)")
    print("=" * 70)
    leakage = False
    for cls in CLASSES:
        hashes = {sp: set() for sp in SPLITS}
        for sp in SPLITS:
            d = OUTPUT_DIR / sp / cls
            if d.exists():
                for f in d.iterdir():
                    if f.is_file():
                        hashes[sp].add(sha256_file(f))
        tv = hashes["train"] & hashes["val"]
        tt = hashes["train"] & hashes["test"]
        vt = hashes["val"]   & hashes["test"]
        if tv or tt or vt:
            leakage = True
            print(f"  FAIL {cls}: tv={len(tv)}, tt={len(tt)}, vt={len(vt)}")
        else:
            print(f"  OK   {cls}: No leakage")

    # ── Summary ───────────────────────────────────────────────────────────────
    print()
    print("=" * 70)
    print("FINAL DATASET SUMMARY")
    print("=" * 70)
    print(f"{'Class':<20} {'Train':>6} {'Val':>6} {'Test':>6} {'Total':>7}  Status")
    print("-" * 60)
    gt = gv = gte = gtotal = 0
    for cls in CLASSES:
        r  = class_results.get(cls, {sp:0 for sp in SPLITS+["total"]})
        ok = (r["train"]==TARGET_TRAIN and r["val"]==TARGET_VAL and r["test"]==TARGET_TEST)
        print(f"{cls:<20} {r['train']:>6} {r['val']:>6} {r['test']:>6} {r['total']:>7}  {'OK' if ok else 'SHORTAGE'}")
        gt += r["train"]; gv += r["val"]; gte += r["test"]; gtotal += r["total"]
    print("-" * 60)
    print(f"{'TOTAL':<20} {gt:>6} {gv:>6} {gte:>6} {gtotal:>7}")
    print(f"\nExpected: {TARGET_TRAIN*5} / {TARGET_VAL*5} / {TARGET_TEST*5} / {TARGET_PER_CLASS*5}")

    # ── Shortage report ───────────────────────────────────────────────────────
    if shortage_report:
        print()
        print("=" * 70)
        print("SHORTAGE REPORT")
        print("=" * 70)
        for cls, info in shortage_report.items():
            print(f"  {cls}: available={info['available']}, shortage={info['shortage']}")
            print(f"    --> Collect {info['shortage']} more verified '{cls}' images and rerun.")
        print()
        print("  *** DO NOT TRAIN -- dataset is IMBALANCED. ***")
    else:
        print("\n  ALL CLASSES BALANCED (500 each). Dataset READY.")

    # ── Validation checks ─────────────────────────────────────────────────────
    print()
    print("=" * 70)
    print("VALIDATION CHECKS")
    print("=" * 70)
    passed = total_c = 0

    def chk(label, cond):
        nonlocal passed, total_c
        total_c += 1
        mark = "PASS" if cond else "FAIL"
        print(f"  [{mark}] {label}")
        if cond: passed += 1

    chk("Grand total correct or shortage reported",
        gtotal == TARGET_PER_CLASS * 5 or bool(shortage_report))
    chk("No cross-split leakage", not leakage)
    chk("All 5 classes processed", len(class_results) == 5)
    for sp in SPLITS:
        for cls in CLASSES:
            d = OUTPUT_DIR / sp / cls
            chk(f"Folder exists: {sp}/{cls}", d.exists())
            if d.exists() and cls not in shortage_report:
                count = len([f for f in d.iterdir() if f.is_file()])
                tgt = TARGET_TRAIN if sp=="train" else TARGET_VAL if sp=="val" else TARGET_TEST
                chk(f"Count correct: {sp}/{cls} ({count}/{tgt})", count == tgt)

    print(f"\n  Checks: {passed}/{total_c} passed")
    print()
    if shortage_report:
        print("INCOMPLETE -- Resolve shortages before training.")
        sys.exit(1)
    else:
        print("COMPLETE -- All checks passed!")
        sys.exit(0)


if __name__ == "__main__":

#!/usr/bin/env python3
# ============================================================================
# PROJECT EXTRACTION UTILITY
# ============================================================================
# Purpose: Generate a comprehensive report of the entire Python codebase
# Output: Single text file containing:
#         1. Visual tree structure of project directories/files
#         2. Complete content of all Python files
#         3. Summary statistics
# Usage: python Project_extract.py
# Use Case: Code review, documentation, sharing codebase overview
# ============================================================================

"""
Extract all Python files with their content and project structure into a single text file.
"""

import os
import sys
from pathlib import Path


def get_project_structure(root_dir, prefix="", max_depth=10, current_depth=0, ignore_dirs=None):
    """
    Generate a text representation of the project structure.
    
    Creates a tree-style visualization of directories and files.
    Recursively traverses subdirectories while respecting depth limits.
    
    Args:
        root_dir: Starting directory path
        prefix: String prefix for tree indentation
        max_depth: Maximum recursion depth to prevent infinite loops
        current_depth: Current recursion level
        ignore_dirs: Set of directory names to skip
    
    Returns:
        String containing formatted directory tree
    """
    if ignore_dirs is None:
        ignore_dirs = {".git", "__pycache__", ".pytest_cache", "*.egg-info", ".venv", "venv", "env", ".csv", "node_modules", "dist", "build", ".expo"}
    
    if current_depth >= max_depth:
        return ""
    
    structure = ""
    try:
        items = sorted(os.listdir(root_dir))
    except PermissionError:
        return ""
    
    dirs = []
    files = []
    
    for item in items:
        # Skip hidden files/dirs and common ignore patterns
        if item.startswith("."):
            continue
        if item in ignore_dirs or any(item.endswith(x.replace("*", "")) for x in ignore_dirs if "*" in x):
            continue
        
        item_path = os.path.join(root_dir, item)
        if os.path.isdir(item_path):
            dirs.append(item)
        else:
            files.append(item)
    
    # Add files first
    for i, file in enumerate(files):
        is_last_file = (i == len(files) - 1) and len(dirs) == 0
        structure += f"{prefix}{'└── ' if is_last_file else '├── '}{file}\n"
    
    # Add directories
    for i, dir_name in enumerate(dirs):
        is_last = i == len(dirs) - 1
        structure += f"{prefix}{'└── ' if is_last else '├── '}{dir_name}/\n"
        
        dir_path = os.path.join(root_dir, dir_name)
        extension = "    " if is_last else "│   "
        structure += get_project_structure(dir_path, prefix + extension, max_depth, current_depth + 1, ignore_dirs)
    
    return structure


def extract_python_files(root_dir, output_file):
    """
    Extract all Python files and their content.
    
    Main extraction function that:
    1. Walks through directory tree to find all .py files
    2. Generates project structure visualization
    3. Writes formatted output with all file contents
    4. Provides summary statistics
    
    Args:
        root_dir: Root directory to scan
        output_file: Path to output text file
    """
    py_files = []
    
    # Collect all Python files
    for root, dirs, files in os.walk(root_dir):
        # Skip certain directories
        dirs[:] = [d for d in dirs if d not in {".git", "__pycache__", ".pytest_cache", ".venv", "venv", "env", ".csv", "node_modules", "dist", "build", ".expo"}]
        
        for file in files:
            if file.endswith(".py"):
                file_path = os.path.join(root, file)
                py_files.append(file_path)
    
    # Sort files for consistent output
    py_files.sort()
    
    # Write to output file
    with open(output_file, "w", encoding="utf-8") as out:
        # Header
        out.write("=" * 80 + "\n")
        out.write("PROJECT EXTRACTION REPORT\n")
        out.write("=" * 80 + "\n\n")
        
        # Project Structure
        out.write("PROJECT STRUCTURE:\n")
        out.write("-" * 80 + "\n\n")
        out.write(f"{os.path.basename(root_dir)}/\n")
        out.write(get_project_structure(root_dir))
        out.write("\n\n")
        
        # File Contents
        out.write("=" * 80 + "\n")
        out.write("FILE CONTENTS\n")
        out.write("=" * 80 + "\n\n")
        
        for file_path in py_files:
            # Calculate relative path
            rel_path = os.path.relpath(file_path, root_dir)
            
            # Write file header
            out.write(f"\n{'=' * 80}\n")
            out.write(f"FILE: {rel_path}\n")
            out.write(f"{'=' * 80}\n\n")
            
            try:
                with open(file_path, "r", encoding="utf-8") as f:
                    content = f.read()
                    out.write(content)
            except Exception as e:
                out.write(f"[ERROR] Could not read file: {str(e)}\n")
            
            out.write("\n")
        
        # Summary
        out.write("\n" + "=" * 80 + "\n")
        out.write(f"SUMMARY: {len(py_files)} Python files extracted\n")
        out.write("=" * 80 + "\n")
    
    print(f"✓ Extraction complete!")
    print(f"✓ Total Python files: {len(py_files)}")
    print(f"✓ Output saved to: {output_file}")
    print(f"✓ File size: {os.path.getsize(output_file) / 1024:.2f} KB")


if __name__ == "__main__":
    # Get project root (current directory)
    project_root = os.getcwd()
    
    # Output file
    output_filename = "project_extraction.txt"
    output_path = os.path.join(project_root, output_filename)
    
    print(f"Extracting Python files from: {project_root}")
    print(f"Output file: {output_path}\n")
    
    extract_python_files(project_root, output_path)

"""
TurSathi AI — Non-Destructive Organ Dataset Staging Script

Copies reusable pigeonpea leaf and non-tur images into the 4-class organ dataset
structure while strictly preserving all original datasets.

SAFETY CONSTRAINTS:
1. Sources are strictly READ-ONLY (uses shutil.copy2).
2. Never modifies, renames, moves, or deletes files in source directories.
3. Preserves established train/val/test split boundaries to prevent data leakage.
4. Leaves Tur_Pod and Tur_WholePlant_or_Canopy directories ready for genuine field collection.
"""

import argparse
import logging
import shutil
import sys
from pathlib import Path

logging.basicConfig(level=logging.INFO, format="%(asctime)s [%(levelname)s] %(message)s")
logger = logging.getLogger("tursathi.organ_stager")

PROJECT_ROOT = Path(__file__).resolve().parent.parent
LEAF_SOURCE = PROJECT_ROOT / "datasets" / "pigeonpea_disease_final"
NOTTUR_SOURCE = PROJECT_ROOT / "datasets" / "tur_image_validator_v3"
DEST_ROOT = PROJECT_ROOT / "datasets" / "pigeonpea_organ_dataset"

# Exclude "Wilt": 100% of Wilt images (287 files) depict whole plants, stems, and distant canopy.
# Only genuine foliar leaf crops are staged as Tur_Leaf.
TRUE_LEAF_CLASSES = ["Healthy", "Leaf_Spot", "Leaf_webber", "Sterilic_mosaic"]
SPLITS = ["train", "val", "test"]


def validate_sources() -> bool:
    if not LEAF_SOURCE.exists():
        logger.error("Leaf source directory missing: %s", LEAF_SOURCE)
        return False
    if not NOTTUR_SOURCE.exists():
        logger.error("Not_Tur source directory missing: %s", NOTTUR_SOURCE)
        return False
    return True


def stage_dataset(dry_run: bool = False):
    logger.info("Starting organ dataset staging (dry_run=%s)", dry_run)
    logger.info("Destination: %s", DEST_ROOT)

    if not validate_sources():
        sys.exit(1)

    copied_leaf = 0
    copied_nottur = 0

    for split in SPLITS:
        leaf_dest = DEST_ROOT / split / "Tur_Leaf"
        nottur_dest = DEST_ROOT / split / "Not_Tur_or_Unsupported"
        pod_dest = DEST_ROOT / split / "Tur_Pod"
        canopy_dest = DEST_ROOT / split / "Tur_WholePlant_or_Canopy"

        if not dry_run:
            leaf_dest.mkdir(parents=True, exist_ok=True)
            nottur_dest.mkdir(parents=True, exist_ok=True)
            pod_dest.mkdir(parents=True, exist_ok=True)
            canopy_dest.mkdir(parents=True, exist_ok=True)

        # 1. Stage genuine Tur_Leaf from pigeonpea_disease_final (excluding Wilt whole-plant photos)
        for disease in TRUE_LEAF_CLASSES:
            src_dir = LEAF_SOURCE / split / disease
            if not src_dir.exists():
                continue
            for img in src_dir.glob("*.*"):
                if img.suffix.lower() not in [".jpg", ".jpeg", ".png", ".webp"]:
                    continue
                target = leaf_dest / f"{disease.lower()}_{img.name}"
                if not target.exists():
                    if not dry_run:
                        shutil.copy2(img, target)
                    copied_leaf += 1

        # 2. Stage Not_Tur_or_Unsupported from tur_image_validator_v3
        src_nottur = NOTTUR_SOURCE / split / "Not_Tur"
        if src_nottur.exists():
            for img in src_nottur.glob("*.*"):
                if img.suffix.lower() not in [".jpg", ".jpeg", ".png", ".webp"]:
                    continue
                target = nottur_dest / img.name
                if not target.exists():
                    if not dry_run:
                        shutil.copy2(img, target)
                    copied_nottur += 1

    print("\n" + "=" * 60)
    print("STAGING SUMMARY:")
    print("=" * 60)
    print(f"  Tur_Leaf images staged                 : {copied_leaf:>5d}")
    print(f"  Not_Tur_or_Unsupported images staged   : {copied_nottur:>5d}")
    print(f"  Tur_Pod images available              :     0 (Awaiting field collection)")
    print(f"  Tur_WholePlant_or_Canopy available     :     0 (Awaiting field collection)")
    print("=" * 60)
    print("STATUS: Original datasets remain untouched and immutable.")


def main():
    parser = argparse.ArgumentParser(description="Stage reusable classes for organ detector")
    parser.add_argument("--execute", action="store_true", help="Execute copy operation (default is dry-run)")
    args = parser.parse_args()

    stage_dataset(dry_run=not args.execute)


if __name__ == "__main__":
    main()
